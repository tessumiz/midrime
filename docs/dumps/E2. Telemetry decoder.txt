cls Telemetry:
    history = [:]
    active_alerts = []

    init(self, cfg): @fill

    get_thr(sensor_type, metric_name):
        sensors    = cfg["sensors"]         ?? [:]
        stype_cfg  = sensors[sensor_type]   ?? [:]
        metric_cfg = stype_cfg[metric_name] ?? [:]

        metric_cfg["limit"] ? 100.0


    get_avg_calc(win_sz):
        (vals) { vals ? avg(vals[-win_sz..]) : 0.0 }


    def proc_batch(evs):
        grp = [:]

        for ev in evs:
            sid = ev["id"] ?? "unknown"

            if sid nin grp:
                grp[sid] = [:]

            grp[sid] = []


        ma_calc = get_avg_calc(5)

        for sid, ev_lst in grp:
            if sid nin history:
                history[sid] = [:]

            reads = ev_lst as e
                => sel e["is_valid"] ?? nil
                => e["value"] ?? 0.0

            history[sid] << reads
            avg = ma_calc(history[sid])

            sensor_type =
                ev_lst ? (ev_lst[0]["type"] ?? "unknown") : "unknown"

            limit = get_thr(sensor_type, "temp")

            if avg > limit:
                active_alerts << [
                    sensor_id: sid,
                    avg: avg,
                    limit: limit
                ]

        act_alerts as alert
            => sel alert["avg"] > alert["limit"] * 1.5
            => print("foo")


cfg = [
    sensors: [
        engine_temp: [ temp: [limit: 120.0] ],
        cabin_temp:  [ temp: [limit: 30.0] ]
    ]
]

aggr = Telemetry(cfg)
inc_ev = ...
aggr.proc_batch(inc_ev)
